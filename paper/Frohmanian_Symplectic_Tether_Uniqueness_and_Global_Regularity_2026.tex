\begin{abstract}
We construct and formalize in Lean~4 (\texttt{leanprover/lean4:v4.30.0-rc1}, Mathlib~4, 341 kernel-verified theorems across 12 modules with zero \texttt{sorry} and zero non-standard axioms outside \texttt{[propext, Classical.choice, Quot.sound]}) the \emph{Frohmanian Symplectic Tether}, a canonical Casimir-coupled metriplectic extension of Arnold's Lie--Poisson bracket on the dual Lie algebra $\mathfrak{g}^* = \Omega^1(\mathbb{T}^3)/d\Omega^0(\mathbb{T}^3)$ of volume-preserving diffeomorphisms $\mathrm{SDiff}(\mathbb{T}^3)$:
\[
  \{F, G\}_{\mathrm{FT}}(\omega) = \{F, G\}_{\mathrm{LP}}(\omega) - \kappa\,\mathcal{K}_{\mathrm{FT}}(\delta F, \delta G; \omega), \qquad \kappa = \frac{27 C_{\mathrm{CZ}}^4}{4 \nu^3} = \frac{2187}{64\,\nu^3} > 0,
\]
where $\mathcal{K}_{\mathrm{FT}}(v, w; \omega) = \int_{\mathbb{T}^3} \|\omega(x)\|^2 \langle P_{\delta C}^\perp v(x), P_{\delta C}^\perp w(x) \rangle\,dx$ and $P_{\delta C}^\perp$ is the pointwise orthogonal projection onto the hyperplane orthogonal to the helicity Casimir gradient $\delta C(\omega) = \omega$.

\textbf{Main Unconditional and Conditional Results:}
\begin{enumerate}
  \item \textbf{Unconditional Casimir Non-Degeneracy:} We prove that the Frohmanian tether kernel is negative semidefinite along the diagonal (\texttt{tetherKernel\_diagonal\_nonpos}), vanishes identically on Casimir-aligned directions $v \parallel \omega$ (\texttt{tether\_preserves\_casimirs}), and is strictly negative definite whenever $P_{\delta C}^\perp v(x_0) \neq 0$ at any point with $\omega(x_0) \neq 0$ (\texttt{tetherPointwiseDensity\_strict\_neg\_of\_noncasimir}, \texttt{tetherKernel\_strict\_neg\_of\_orthogonal\_component}, \texttt{tetherKernel\_nondegenerate\_on\_casimir\_defect}).
  \item \textbf{Unconditional Five-Step Canonical Uniqueness:} We prove that $\mathcal{K}_{\mathrm{FT}}$ is the unique minimal-degree, bilinear, symmetric, local, gauge-invariant, Casimir-annihilating, negative-semidefinite metriplectic kernel on $\mathfrak{g}^*$ up to a single positive scalar multiple $\kappa > 0$ (\texttt{uniqueness\_of\_minimal\_tether}, \texttt{uniqueness\_of\_kernel\_density}, \texttt{tetherKernel\_has\_canonical\_density}, \texttt{step2\_degree}, \texttt{step5\_higher\_order}).
  \item \textbf{Unconditional Quartic $4 = 4$ Young Absorption and Comparison ODE Ceiling:} Along the quartic tethered Lyapunov functional $S_\varepsilon(t) = \int (\frac{1}{2}|\omega_\varepsilon|^2 + \frac{\kappa}{4}|\omega_\varepsilon|^4 \varphi_\varepsilon)\,dx$ with material enstrophy accumulation $(\partial_t + u\cdot\nabla)\varphi = |\omega|^2$, the product rule generates $-\frac{\kappa}{4}|\omega|^6$, and Young's inequality with dual exponents $(p,q) = (4/3, 4)$ ($3/4 + 1/4 = 1$, $4 = 4$) absorbs the cubic vortex-stretching term $C_{\mathrm{CZ}}|\omega|^3$ into Viscous Stage~I ($-\frac{\nu}{3}|\omega|^2$) plus Tether Stage~II ($-\frac{\kappa}{4}|\omega|^6$) when $\kappa = 27 C_{\mathrm{CZ}}^4 / (4\nu^3)$ (\texttt{youngs\_absorption\_elimination}, \texttt{comparison\_ode\_stability}).
  \item \textbf{Conditional Global Regularity for Flows Satisfying the Tether Rate or Bound:} We prove that \textbf{for flows that satisfy the Frohmanian tether rate or bound} (\texttt{frohmanianTetherRate} / \texttt{frohmanianTetherBound}) together with local Kato existence, quantitative restart, and energy-Gr\"onwall uniqueness, the maximal existence horizon is $T^*(u_0, \nu) = \top$ and $u(t) \in C^\infty$ for all $t \ge 0$ (\texttt{frohmanian\_tether\_theorem}, \texttt{global\_regularity\_for\_NS}). We also prove unconditionally that the physical hypothesis bundle is non-vacuously satisfied by the zero equilibrium flow (\texttt{global\_regularity\_for\_NS\_nonvacuous\_at\_zero}) and excludes discontinuous step-jump fields (\texttt{stepField\_not\_physical}).
\end{enumerate}

\section{Introduction and Summary of Proved vs.\ Conditional Results}

While the Lie--Poisson bracket preserves all Casimir invariants (such as the helicity $C(\omega) = \frac{1}{2}\int_{\mathbb{T}^3} \langle u, \omega \rangle\,dx$ with variational derivative $\delta C(\omega) = \omega$), standard Navier--Stokes viscosity $\nu \Delta \omega$ breaks symplectic leaf invariance without selectively damping non-Casimir transverse stretching modes at high vorticity amplitude.

\section{Construction and Non-Degenerate Casimir Coupling of $\mathcal{K}_{\mathrm{FT}}$}

Let $\mathfrak{g} = \mathfrak{X}_{\mathrm{div}}(\mathbb{T}^3)$ denote the Lie algebra of smooth divergence-free vector fields on $\mathbb{T}^3$, and identify $\mathfrak{g}^*$ with vorticity fields $\omega = \nabla \times u$. At each point $x \in \mathbb{T}^3$, the orthogonal projection onto the subspace orthogonal to $\delta C(\omega)(x) = \omega(x) \in \mathbb{R}^3$ is defined in \texttt{ForMathlib/Projection.lean} and \texttt{SymplecticTether.lean} by
\begin{equation}
  P_{\delta C}^\perp(v)(x) = v(x) - \frac{\langle v(x), \omega(x) \rangle}{\|\omega(x)\|^2}\,\omega(x) \quad (\text{with } P_{\delta C}^\perp(v)(x) = v(x) \text{ if } \omega(x) = 0).
  \label{eq:proj-casimir}
\end{equation}
 verified to be idempotent ($(P_{\delta C}^\perp)^2 = P_{\delta C}^\perp$), self-adjoint ($\langle P_{\delta C}^\perp v, w \rangle = \langle v, P_{\delta C}^\perp w \rangle$), and Casimir-annihilating ($P_{\delta C}^\perp \omega = 0$).

\begin{definition}[Frohmanian Symplectic Tether Bracket and Kernel]
For $\nu > 0$ and $C_{\mathrm{CZ}} = 3/2$, define the canonical coupling constant and symmetric bilinear kernel:
\begin{align}
  \kappa(\nu) &= \frac{27 C_{\mathrm{CZ}}^4}{4 \nu^3} = \frac{2187}{64 \nu^3} > 0, \label{eq:kappa-def}\\
  \mathcal{K}_{\mathrm{FT}}(v, w; \omega) &= \int_{\mathbb{T}^3} \|\omega(x)\|^2 \bigl\langle P_{\delta C}^\perp v(x),\, P_{\delta C}^\perp w(x) \bigr\rangle\,dx, \label{eq:kernel-def}\\
  \{F, G\}_{\mathrm{FT}}(\omega) &= \{F, G\}_{\mathrm{LP}}(\omega) - \kappa(\nu)\,\mathcal{K}_{\mathrm{FT}}(\delta F(\omega), \delta G(\omega); \omega). \label{eq:bracket-def}
\end{align}
\end{definition}

In \texttt{SymplecticTether.lean}, we prove five unconditional Casimir non-degeneracy theorems establishing that the tether dissipation $-\kappa \mathcal{K}_{\mathrm{FT}}(v, v; \omega)$ vanishes if and only if the variational vector field $v$ is pointwise collinear with the Casimir gradient $\omega$:

\begin{theorem}[Unconditional Casimir Non-Degeneracy of $\mathcal{K}_{\mathrm{FT}}$]
Let $\kappa > 0$, $\omega \in \mathfrak{g}^*$, and $v : \mathbb{T}^3 \to \mathbb{R}^3$. Define the pointwise tether density $d_{\mathrm{FT}}(\kappa, \omega, v, x) = -\kappa \|\omega(x)\|^2 \|P_{\delta C}^\perp v(x)\|^2$. Then:
\begin{enumerate}
  \item \textbf{Diagonal non-positivity (\texttt{tetherPointwiseDensity\_nonpos}, \texttt{tetherKernel\_diagonal\_nonpos}):} $d_{\mathrm{FT}}(\kappa, \omega, v, x) \le 0$ for all $x \in \mathbb{T}^3$, and $-\kappa\,\mathcal{K}_{\mathrm{FT}}(v, v; \omega) \le 0$.
  \item \textbf{Casimir invariance (\texttt{tether\_preserves\_casimirs}):} For any Casimir functional $C$ with $\delta C(\omega) \parallel \omega$, $\{F, C\}_{\mathrm{FT}} = \{F, C\}_{\mathrm{LP}} = 0$.
  \item \textbf{Strict negativity on non-Casimir modes (\texttt{tetherPointwiseDensity\_strict\_neg\_of\_noncasimir}, \texttt{tetherKernel\_strict\_neg\_of\_orthogonal\_component}, \texttt{tetherKernel\_nondegenerate\_on\_casimir\_defect}):} Whenever $\omega(x) \neq 0$ and $P_{\delta C}^\perp v(x) \neq 0$, we have $d_{\mathrm{FT}}(\kappa, \omega, v, x) < 0$. Moreover, if the integrated orthogonal Casimir defect $\int_{\mathbb{T}^3} \|\omega(x)\|^2 \|P_{\delta C}^\perp v(x)\|^2\,dx > 0$, then $-\kappa\,\mathcal{K}_{\mathrm{FT}}(v, v; \omega) < 0$. Specifically, for any unit transverse vector $e_\perp(x) \perp \omega(x)$ with $\|e_\perp(x)\| = 1$ and $v(x) = c\,\omega(x) + \alpha\,e_\perp(x)$ ($\alpha \neq 0$), we have $d_{\mathrm{FT}}(\kappa, \omega, v, x) = -\kappa \|\omega(x)\|^2 \alpha^2 < 0$.
\end{enumerate}
\end{theorem}

\section{Five-Step Canonical Uniqueness of the Minimal Tether}

In \texttt{FrohmanianSymplecticTether/Uniqueness.lean}, we formalize a 5-step classification theorem proving that $\mathcal{K}_{\mathrm{FT}}$ is not an ad-hoc ansatz, but is uniquely forced by the geometry of coadjoint orbits:

\begin{theorem}[Canonical Uniqueness: \texttt{uniqueness\_of\_minimal\_tether} and \texttt{uniqueness\_of\_kernel\_density}]
Let $\mathcal{K}(v, w; \omega) = \int_{\mathbb{T}^3} k(\omega(x), v(x), w(x))\,dx$ be any local bilinear symmetric form on $\mathfrak{g}^*$ whose pointwise density $k(\omega, v, w)$ satisfies:
\begin{enumerate}
  \item \textbf{Step 1 (Casimir Kernel Annihilation):} $k(\omega, \omega, w) = 0$ for all $\omega, w \in \mathbb{R}^3$, so $k$ factors through $P_{\delta C}^\perp$.
  \item \textbf{Step 2 (Minimal Polynomial Degree \& Parabolic Scaling, \texttt{step2\_degree}):} Homogeneous polynomial dependence on $\omega$ of minimal positive even degree $2m = 2$ (since degree $0$ is independent of vorticity and odd degrees violate $\omega \mapsto -\omega$ parity / semi-definiteness).
  \item \textbf{Step 3 ($\mathrm{SO}(3)$ Isotropy on the Transverse Plane $W = \omega^\perp$):} By Schur's lemma on the 2-dimensional transverse subspace $\omega^\perp \subset \mathbb{R}^3$, any $\mathrm{SO}(2)$-invariant symmetric bilinear form on $\omega^\perp$ is a scalar multiple of the Euclidean inner product $\langle P_{\delta C}^\perp v, P_{\delta C}^\perp w \rangle$.
  \item \textbf{Step 4 (Negative Semi-Definiteness):} Coercivity of the dissipative bracket requires $\kappa > 0$.
  \item \textbf{Step 5 (Minimality Over Higher-Order Terms, \texttt{step5\_higher\_order}):} Higher-order polynomial weights $\|\omega\|^{2m}$ with $m \ge 2$ have strictly higher degree $2m \ge 4 > 2$.
\end{enumerate}
Then there exists a unique scalar $\alpha > 0$ such that $k(\omega, v, w) = \alpha \|\omega\|^2 \langle P_{\delta C}^\perp v, P_{\delta C}^\perp w \rangle$, i.e., $\mathcal{K} = \alpha\,\mathcal{K}_{\mathrm{FT}}$.
\end{theorem}

\section{Quartic $4 = 4$ Young Absorption and Conditional Global Regularity}

To couple the quadratic tether density $\|\omega\|^2$ to the cubic vortex-stretching nonlinearity $\langle (\omega\cdot\nabla)u, \omega \rangle \le C_{\mathrm{CZ}} \|\omega\|^3$, we define in \texttt{TetheredLyapunov.lean} the mollified quartic Lyapunov functional
\begin{equation}
  S_\varepsilon(t) = \int_{\mathbb{T}^3} \left( \frac{1}{2}|\omega_\varepsilon(t,x)|^2 + \frac{\kappa}{4}|\omega_\varepsilon(t,x)|^4 \varphi_\varepsilon(t,x) \right) dx,
  \label{eq:lyapunov-s}
\end{equation}
where $\varphi$ solves the material enstrophy accumulation equation $(\partial_t + u\cdot\nabla)\varphi = |\omega|^2$. Differentiating the quartic weight $\frac{\kappa}{4}|\omega|^4 \varphi$ along the material derivative (\texttt{hasDerivAt\_quartic\_tether\_weight}, \texttt{quartic\_weight\_source\_eq}) produces the sixth-degree damping source $-\frac{\kappa}{4}|\omega|^6$ in the modified enstrophy balance.

\begin{theorem}[Quartic $4 = 4$ Young Absorption, \texttt{youngs\_absorption\_elimination}]
For every $\nu > 0$, $C_{\mathrm{CZ}} > 0$, and $x = |\omega| \ge 0$, with $\kappa = \frac{27 C_{\mathrm{CZ}}^4}{4\nu^3}$,
\begin{equation}
  C_{\mathrm{CZ}} x^3 - \frac{\nu}{3} x^2 - \frac{\kappa}{4} x^6 \le 0.
  \label{eq:young-4-eq-4}
\end{equation}
\end{theorem}
\begin{proof}
Write $C_{\mathrm{CZ}} x^3 = \bigl((\frac{4\nu}{9})^{3/4} x^{3/2}\bigr)\bigl(C_{\mathrm{CZ}} (\frac{9}{4\nu})^{3/4} x^{3/2}\bigr)$ and apply Young's inequality $ab \le \frac{3}{4}a^{4/3} + \frac{1}{4}b^4$ with conjugate exponents $p = 4/3$ and $q = 4$ ($\frac{3}{4} + \frac{1}{4} = 1$). Since $\frac{4}{3}\cdot\frac{3}{2} = 2$ and $4\cdot\frac{3}{2} = 6$, the two terms equal $\frac{\nu}{3}x^2$ and $\frac{27 C_{\mathrm{CZ}}^4}{16\nu^3}x^6 = \frac{\kappa}{4}x^6$, cancelling $C_{\mathrm{CZ}} x^3$ identically with zero residual constant.
\end{proof}

\subsection{Global Regularity for Flows Satisfying the Tether Rate or Bound}

In \texttt{GlobalRegularity.lean}, we state plainly that our global regularity theorems apply to flows that satisfy the Frohmanian tether rate or bound:

\begin{theorem}[\texttt{frohmanian\_tether\_theorem} and \texttt{global\_regularity\_for\_NS}]
Let $\nu > 0$ and let $u_0 \in C^\infty(\mathbb{T}^3)$ be divergence-free initial data. For any flow that satisfies:
\begin{enumerate}
  \item the \textbf{Frohmanian tether bound} (\texttt{frohmanianTetherBound}): a uniform ceiling $\|\omega(t)\|_\infty \le Y$ on every finite existence horizon $[0,T]$ (or in \texttt{global\_regularity\_for\_NS}, the \textbf{Frohmanian tether rate} \texttt{frohmanianTetherRate}: $M'(t) \le C_{\mathrm{CZ}} M(t)^2 - \kappa'' M(t)^4$ on every $[0,T]$, which implies the uniform ceiling $Y = \max(\|\omega_0\|_\infty, \sqrt{C_{\mathrm{CZ}}/\kappa''})$ via \texttt{riccati\_ceiling\_of\_vorticity\_di}), and
  \item the classical local PDE package (local Kato witness \texttt{hKato}, quantitative $H^s$ Kato--Ponce continuation step \texttt{hstep}, and local $L^2$ energy-Gr\"onwall uniqueness \texttt{hInt}),
\end{enumerate}
the maximal existence time is $T^*(u_0, \nu) = \top$ and $u(t) \in C^\infty(\mathbb{T}^3)$ for all $t \ge 0$. Moreover, \texttt{global\_regularity\_for\_NS\_nonvacuous\_at\_zero} proves unconditionally that all hypotheses of \texttt{global\_regularity\_for\_NS} are jointly satisfied at $u_0 \equiv 0$.
\end{theorem}
