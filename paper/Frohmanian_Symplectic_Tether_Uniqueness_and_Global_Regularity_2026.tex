\begin{abstract}
\textbf{Conditional result. Not a Clay Millennium claim.} Nothing here proves global regularity for the 3D Navier--Stokes equations.

We formalize in Lean~4 (\texttt{leanprover/lean4:v4.30.0-rc1}, Mathlib commit \texttt{0c154d6}; 531 \texttt{theorem}/\texttt{lemma} declarations in the 20 modules of the \texttt{FrohmanianSymplecticTether} library, comment-aware source count; no \texttt{sorry}; only the axioms \texttt{propext}, \texttt{Classical.choice}, \texttt{Quot.sound}) the \emph{Frohmanian Symplectic Tether}, a Casimir-coupled metriplectic extension of Arnold's Lie--Poisson bracket on the dual Lie algebra of volume-preserving diffeomorphisms:
\[
  \{F, G\}_{\mathrm{FT}}(\omega) = \{F, G\}_{\mathrm{LP}}(\omega) - \kappa\,\mathcal{K}_{\mathrm{FT}}(\delta F, \delta G; \omega), \qquad \kappa > 0,
\]
where $\mathcal{K}_{\mathrm{FT}}(v, w; \omega) = \int \|\omega(x)\|^2 \langle P_{\delta C}^\perp v(x), P_{\delta C}^\perp w(x) \rangle\,dx$ and $P_{\delta C}^\perp$ is the pointwise orthogonal projection onto the hyperplane orthogonal to the helicity Casimir gradient $\delta C(\omega) = \omega$. In the Lean development $\kappa$ is a positive parameter; the value $\kappa=27C_{\mathrm{CZ}}^4/(4\nu^3)$ used as motivation below is not formalized.

The structural results are packaged into the 4-conjunct product theorem (its only hypothesis is $0<\kappa$) \texttt{frohmanian\_metriplectic\_system\_canonical\_uniqueness\_and\_global\_regularity\_product\_theorem} ($P_1 \wedge P_2 \wedge P_3 \wedge P_4$, Section~\ref{sec:product}). $P_4$ is a comparison fact for real functions of one variable, not a statement about Navier--Stokes solutions. The Navier--Stokes theorem \texttt{global\_regularity\_for\_NS} is \emph{conditional}: it assumes the open time-uniform vorticity bound \texttt{FrohmanianTetherBound} together with the local-theory hypotheses \texttt{KatoLocalWitness} (local existence), \texttt{PhysicalBKMContinuationCriterionFor} (Beale--Kato--Majda continuation) and \texttt{KatoUniquenessOnExistenceTimes} (uniqueness); these are discharged only at $u_0\equiv0$.

\textbf{Model caveat.} The Lean type \texttt{T3} is \texttt{EuclideanSpace}~$\mathbb{R}$~\texttt{(Fin 3)}, i.e.\ $\mathbb{R}^3$ with Lebesgue measure, not the torus. \texttt{global\_regularity\_for\_NS} assumes $\int_{\mathbb{R}^3}\|u_0\|^2<\infty$, which a nonzero $\mathbb{Z}^3$-periodic field never satisfies, so this theorem does not model genuine periodic data.
\end{abstract}

\section{Introduction and Single-Sentence Final Product Form ($P_1 \wedge P_2 \wedge P_3 \wedge P_4$)}\label{sec:product}

In Arnold's geometric formulation of ideal incompressible hydrodynamics \cite{Arnold1966}, Euler's equation on the 3-torus $\mathbb{T}^3$ is a Hamiltonian flow on dual coadjoint orbits of the volume-preserving diffeomorphism group $\mathrm{SDiff}(\mathbb{T}^3)$ with respect to the Kirillov--Kostant--Souriau / Lie--Poisson bracket $\{F, G\}_{\mathrm{LP}}(\omega) = \int_{\mathbb{T}^3} \langle \omega, [\delta F, \delta G] \rangle\,dx$. In \texttt{FrohmanianSymplecticTether/FinalProductTheorem.lean}, we prove the master 4-conjunct product theorem packaging the canonical 5-step uniqueness, metriplectic Casimir coupling, quartic $4=4$ Young absorption, and Lyapunov ODE ceiling into a single theorem statement.

\begin{theorem}[Master Product Theorem: \texttt{frohmanian\_metriplectic\_system\_canonical\_uniqueness\_and\_global\_regularity\_product\_theorem}]
For every $\kappa>0$ the following four statements hold (exact Lean definitions in \texttt{FinalProductTheorem.lean}; Listing~\ref{lst:product}).
\textbf{(P$_1$, \texttt{P1\_CanonicalFiveStepUniqueness})} Every symmetric bilinear correction $B$ satisfying coadjoint invariance (C1), degeneracy with respect to kinetic energy (C2), controllable negative feedback (C3), quadratic saturation and polarization, or having the canonical kernel density $-\kappa\|\omega(x)\|^2$, equals \texttt{TetherKernel}~$\kappa$; and \texttt{TetherKernel} scales by $c^2$ under $\omega\mapsto c\omega$ ($c\neq0$) when the functional derivatives are scale-invariant.
\textbf{(P$_2$, \texttt{P2\_MetriplecticSplitAndCasimirDamping})} The KKS 2-form is skew and satisfies the pointwise cyclic Jacobi identity; the Frohmanian pairing on the diagonal equals the tether metric term and is $\le0$; the tethered bracket reproduces the classical bracket on the kinetic-energy Hamiltonian under the stated hypotheses; $\Pi_u(\lambda u)=0$ when $\int\|u\|^2\neq0$; and the pointwise tether density is strictly negative where $\omega(x)\neq0$ and the projected functional derivative is nonzero.
\textbf{(P$_3$, \texttt{P3\_QuarticYoungAbsorption})} If $I_6\ge0$ and $4C_{\mathrm{CZ}}MI_4\le\frac{\kappa}{4}I_6+C_{\mathrm{abs}}M^3\|\phi\|^{3/2}$, then $4C_{\mathrm{CZ}}MI_4-\kappa I_6\le C_{\mathrm{abs}}M^3\|\phi\|^{3/2}-\frac34\kappa I_6$; an attained spatial maximum pins \texttt{vorticity\_sup\_norm}; and the concrete 3D Bochner--polar inequality and the near/far cutoff inequality hold as algebraic statements about real numbers and vectors.
\textbf{(P$_4$, \texttt{P4\_UnconditionalLyapunovCeiling})} For $C,\kappa''>0$, every $y:\mathbb{R}\to\mathbb{R}$ differentiable at every $s\ge0$ with $y'(s)\le Cy(s)^2-\kappa''y(s)^3$ for $s\ge0$ satisfies $y(t)\le\max(y(0),C/\kappa'')$ for all $t\ge0$. This is a scalar comparison fact; that the vorticity of a Navier--Stokes flow obeys this inequality is not proved.

Global regularity for general data, \texttt{global\_regularity\_for\_NS}, is conditional on \texttt{FrohmanianTetherBound} (open), \texttt{KatoLocalWitness}, \texttt{PhysicalBKMContinuationCriterionFor} and \texttt{KatoUniquenessOnExistenceTimes}.

\paragraph{Status (2026-10-30).} The master theorem proves $P_1\wedge P_2\wedge P_3\wedge P_4$ with no hypotheses, no \texttt{sorry}, and only standard axioms. $P_1$ is the uniqueness of the minimal tether kernel. $P_2$ is the metriplectic structure. The KKS part is skew and satisfies Jacobi (\texttt{jacF\_eq\_zero}). The tether is symmetric (\texttt{tetF\_symm}), degenerate in the energy direction (\texttt{tetF\_energy\_left}), and nonpositive on all real fields (\texttt{dissF\_nonpos}). The full bracket is metriplectic, not Lie--Poisson, and fails Jacobi: the mixed cyclic sum has coefficient $\tfrac32\kappa w\neq0$ at $u=(\sin x,0,0)$, and $\tfrac{3i}{8}\kappa w\neq0$ at $u=(\sin y,0,0)$ with the triple $(0,0,\sin x)$. \texttt{tethered\_jacobi\_identity\_conditional} derives Jacobi from its hypotheses only. $P_3$ consists of algebraic and pointwise absorption inequalities. $P_4$ is a comparison fact for real functions satisfying $y'\le Cy^2-\kappa''y^3$; that the vorticity of a flow obeys this inequality is not proved. General global regularity, \texttt{global\_regularity\_for\_NS}, is conditional on \texttt{KatoLocalWitness}, \texttt{FrohmanianTetherBound} (an open, time-uniform $\|\omega\|_\infty$ bound, used only through its finite-horizon consequence \texttt{PhysicalFiniteTimeVorticityBound}), the Beale--Kato--Majda continuation criterion \texttt{PhysicalBKMContinuationCriterionFor}, and local uniqueness \texttt{KatoUniquenessOnExistenceTimes}. $T^\ast=\infty$ is derived from $T^\ast>0$, the criterion and the integrability. These are discharged only at zero data.

\paragraph{Finite-horizon vorticity control.} Clay statement (B) asks for a smooth solution on $[0,\infty)$ with bounded energy; it does not ask for a bound on $\|\omega(t)\|_\infty$ uniform in $t$. The named open hypotheses in the finite-horizon form are
\[
\texttt{FiniteTimeVorticityBound}(u_0,\nu):\quad \forall\,(u,p)\ \text{solving NS from } u_0,\ \ \forall T\ge0\ \exists C(T):\ \sup_{t\in[0,T]}\|\omega(t)\|_\infty\le C(T),\ \ t\mapsto\|\omega(t)\|_\infty \text{ continuous},
\]
\[
\texttt{BKMIntegrability}(u_0,\nu):\quad \forall\,(u,p)\ \text{solving NS from } u_0,\ \ \forall T:\ \int_0^T\|\omega(t)\|_\infty\,dt<\infty,
\]
with physical-path versions \texttt{PhysicalFiniteTimeVorticityBound} and \texttt{PhysicalBKMIntegrability}. Proved: uniform ceiling $\Rightarrow$ finite-horizon bound (take $C(T)=Y$), and finite-horizon bound $\Rightarrow$ BKM integrability (a function continuous on $\mathbb{R}$ is integrable on each compact $[0,T]$). The Beale--Kato--Majda step \texttt{beale\_kato\_majda} takes $T^\ast>0$ (\texttt{LocalExistenceTime}), the continuation criterion (\texttt{BKMContinuationCriterion}: finite $T^\ast$ forces $\int_0^{T^\ast}\|\omega\|_\infty=\infty$), integrability on every $[0,T]$ (\texttt{PathBKMIntegrable}) and smoothness below $T^\ast$ (\texttt{SmoothBelowTstar}); $T^\ast=\infty$ is derived (\texttt{Tstar\_eq\_top\_of\_bkm}), and the integrability is necessary (\texttt{bkm\_criterion\_without\_integrability\_not\_top}). \texttt{global\_regularity}, \texttt{global\_regularity\_of\_bkmIntegrability}, \texttt{global\_regularity\_of\_bkm\_uniqueness} and \texttt{global\_regularity\_for\_NS} go through this step. The criterion and smoothness below $T^\ast$ are classical and not formalized here. \texttt{beale\_kato\_majda\_scaffold} assumes $T^\ast=\infty$ and does not use integrability (\texttt{beale\_kato\_majda\_scaffold\_integrability\_unused}); \texttt{global\_regularity\_of\_restart}, \texttt{global\_regularity\_of\_kato\_estimates} and \texttt{continuation\_under\_global\_existence} derive or assume $T^\ast=\infty$ without the vorticity input. No result here proves global regularity for plain 3D Navier--Stokes or solves the Clay problem.

\section{Construction and Non-Degenerate Casimir Coupling of $\mathcal{K}_{\mathrm{FT}}$}

Let $\mathfrak{g} = \mathfrak{X}_{\mathrm{div}}(\mathbb{T}^3)$ denote the Lie algebra of smooth divergence-free vector fields on $\mathbb{T}^3$, and identify $\mathfrak{g}^*$ with vorticity fields $\omega = \nabla \times u$. At each point $x \in \mathbb{T}^3$, the orthogonal projection onto the subspace orthogonal to $\delta C(\omega)(x) = \omega(x) \in \mathbb{R}^3$ is defined in \texttt{ForMathlib/Projection.lean} and \texttt{SymplecticTether.lean} by
\begin{equation}
  P_{\delta C}^\perp(v)(x) = v(x) - \frac{\langle v(x), \omega(x) \rangle}{\|\omega(x)\|^2}\,\omega(x) \quad (\text{with } P_{\delta C}^\perp(v)(x) = v(x) \text{ if } \omega(x) = 0).
  \label{eq:proj-casimir}
\end{equation}
It is verified to be idempotent ($(P_{\delta C}^\perp)^2 = P_{\delta C}^\perp$), self-adjoint ($\langle P_{\delta C}^\perp v, w \rangle = \langle v, P_{\delta C}^\perp w \rangle$), and Casimir-annihilating ($P_{\delta C}^\perp \omega = 0$).

\begin{definition}[Frohmanian Symplectic Tether Bracket and Kernel]
For $\nu > 0$ and $C_{\mathrm{CZ}} = 3/2$, define the canonical coupling constant and symmetric bilinear kernel:
\begin{align}
  \kappa(\nu) &= \frac{27 C_{\mathrm{CZ}}^4}{4 \nu^3} = \frac{2187}{64 \nu^3} > 0, \label{eq:kappa-def}\\
  \mathcal{K}_{\mathrm{FT}}(v, w; \omega) &= \int_{\mathbb{T}^3} \|\omega(x)\|^2 \bigl\langle P_{\delta C}^\perp v(x),\, P_{\delta C}^\perp w(x) \bigr\rangle\,dx, \label{eq:kernel-def}\\
  \{F, G\}_{\mathrm{FT}}(\omega) &= \{F, G\}_{\mathrm{LP}}(\omega) - \kappa(\nu)\,\mathcal{K}_{\mathrm{FT}}(\delta F(\omega), \delta G(\omega); \omega). \label{eq:bracket-def}
\end{align}
\end{definition}
In the Lean development $\kappa$ is a positive parameter; the value \eqref{eq:kappa-def} is motivation and is not formalized.


In \texttt{SymplecticTether.lean}, we prove five unconditional Casimir non-degeneracy theorems establishing that the tether dissipation $-\kappa \mathcal{K}_{\mathrm{FT}}(v, v; \omega)$ vanishes if and only if the variational vector field $v$ is pointwise collinear with the Casimir gradient $\omega$:

\begin{theorem}[Unconditional Casimir Non-Degeneracy of $\mathcal{K}_{\mathrm{FT}}$]
Let $\kappa > 0$, $\omega \in \mathfrak{g}^*$, and $v : \mathbb{T}^3 \to \mathbb{R}^3$. Define the pointwise tether density $d_{\mathrm{FT}}(\kappa, \omega, v, x) = -\kappa \|\omega(x)\|^2 \|P_{\delta C}^\perp v(x)\|^2$. Then:
\begin{enumerate}
  \item \textbf{Diagonal non-positivity (\texttt{tetherPointwiseDensity\_nonpos}, \texttt{tetherKernel\_diagonal\_nonpos}):} $d_{\mathrm{FT}}(\kappa, \omega, v, x) \le 0$ for all $x \in \mathbb{T}^3$, and $-\kappa\,\mathcal{K}_{\mathrm{FT}}(v, v; \omega) \le 0$.
  \item \textbf{Casimir invariance (\texttt{tether\_preserves\_casimirs}):} For any Casimir functional $C$ with $\delta C(\omega) \parallel \omega$, $\{F, C\}_{\mathrm{FT}} = \{F, C\}_{\mathrm{LP}} = 0$.
  \item \textbf{Strict negativity on non-Casimir modes (\texttt{tetherPointwiseDensity\_strict\_neg\_of\_noncasimir}, \texttt{tetherKernel\_strict\_neg\_of\_orthogonal\_component}, \texttt{tetherKernel\_nondegenerate\_on\_casimir\_defect}):} Whenever $\omega(x) \neq 0$ and $P_{\delta C}^\perp v(x) \neq 0$, we have $d_{\mathrm{FT}}(\kappa, \omega, v, x) < 0$. Moreover, if the integrated orthogonal Casimir defect $\int_{\mathbb{T}^3} \|\omega(x)\|^2 \|P_{\delta C}^\perp v(x)\|^2\,dx > 0$, then $-\kappa\,\mathcal{K}_{\mathrm{FT}}(v, v; \omega) < 0$.
\end{enumerate}
\end{theorem}

\section{Five-Step Canonical Uniqueness of the Minimal Tether}

In \texttt{FrohmanianSymplecticTether/Uniqueness.lean}, we formalize a 5-step classification theorem proving that $\mathcal{K}_{\mathrm{FT}}$ is uniquely forced by the geometry of coadjoint orbits:

\begin{theorem}[Canonical Uniqueness: \texttt{uniqueness\_of\_minimal\_tether} and \texttt{uniqueness\_of\_kernel\_density}]
Let $\mathcal{K}(v, w; \omega) = \int_{\mathbb{T}^3} k(\omega(x), v(x), w(x))\,dx$ be any local bilinear symmetric form on $\mathfrak{g}^*$ whose pointwise density $k(\omega, v, w)$ satisfies C1 (Casimir Kernel Annihilation), C2 (Minimal Polynomial Degree $2m=2$ \& Parabolic Scaling), and C3 ($\mathrm{SO}(3)$ Isotropy \& Negative Semi-Definiteness). Then $B = \texttt{TetherKernel}~\kappa$ (Lean: \texttt{uniqueness\_of\_minimal\_tether}, with $\kappa>0$ a parameter and C1--C3, saturation and polarization as explicit hypotheses).
\end{theorem}

\section{Quartic $4 = 4$ Young Absorption}

\begin{theorem}[Young absorption step, \texttt{youngs\_absorption\_elimination}]
Let $C_{\mathrm{CZ}}=\texttt{CalderonZygmundConstant3D}$ and $\kappa\in\mathbb{R}$. For real $M, I_4, I_6, \|\phi\|, C_{\mathrm{abs}}$ with $I_6\ge0$, if $4C_{\mathrm{CZ}}MI_4\le\frac{\kappa}{4}I_6+C_{\mathrm{abs}}M^3\|\phi\|^{3/2}$ then
\begin{equation}
  4C_{\mathrm{CZ}}MI_4-\kappa I_6\le C_{\mathrm{abs}}M^3\|\phi\|^{3/2}-\tfrac34\kappa I_6.
  \label{eq:young-4-eq-4}
\end{equation}
The Young bound itself is a hypothesis; no Navier--Stokes estimate is derived from it here.
\end{theorem}
